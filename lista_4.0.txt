zad.1

SELECT
    kategoria
FROM
    gry
GROUP BY
    kategoria

zad.2

SELECT
    ocena AS [różne oceny]
FROM
    oceny
GROUP BY
    ocena

zad.3

SELECT
    nazwisko AS [unikalne_nazwiska]
FROM
    gracze
GROUP BY
    nazwisko
ORDER BY
    nazwisko ASC

zad.4

SELECT
    nazwisko + " " + imie AS [unikalne_nazwiska]
FROM
    gracze
GROUP BY
    nazwisko,
    imie
ORDER BY
    nazwisko,
    imie ASC

zad.5

SELECT
    kategoria,
    COUNT(*)
FROM
    gry
GROUP BY
    kategoria

zad.6

SELECT
    id_gry,
    MIN(ocena) AS [min_ocena]
FROM
    oceny
GROUP BY
    id_gry

zad.7

SELECT
    imie,
    count(*) AS [ilosc]
FROM
    gracze
GROUP BY
    imie
ORDER BY
    count(*) DESC

zad.8

SELECT
    stan,
    ROUND(AVG(ocena,), 2) AS [srednia ocen]
FROM
    oceny
GROUP BY
    stan
ORDER BY
    ROUND(AVG(ocena,), 2) ASC

zad.9

SELECT
    id_gry,
    stan,
    ocena,
    COUNT(*) AS [ilosc unikalnych opini]
FROM
    oceny
GROUP BY
    id_gry,
    stan,
    ocena
ORDER BY
    COUNT(*) DESC
